cmake_minimum_required(VERSION 3.10)
project(postgres_catalog_resources LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
find_package(Threads REQUIRED)
enable_testing()
add_executable(postgres_catalog_cache_test catalog_cache_test.cpp)
target_include_directories(postgres_catalog_cache_test PRIVATE ../../src/include)
target_link_libraries(postgres_catalog_cache_test PRIVATE Threads::Threads)
add_test(NAME postgres_catalog_cache_resources COMMAND postgres_catalog_cache_test)
add_executable(postgres_numeric_key_test numeric_key_test.cpp)
target_include_directories(postgres_numeric_key_test PRIVATE ../../src/include)
add_test(NAME postgres_numeric_keys COMMAND postgres_numeric_key_test)
add_executable(postgres_numeric_text_test numeric_text_test.cpp)
